package diagnostics
import (
 "context"
 "errors"
 "testing"
 "math"
 "time"
)
func TestRootFinalLossUsesEndOffset(t *testing.T) {
 c,e:=NewCapture("segment"); if e!=nil {t.Fatal(e)}
 for range CaptureRecordCount {if _,e=c.Offer(Stdout,[]byte("x"));e!=nil {t.Fatal(e)}}
 r,e:=c.Offer(Stdout,[]byte("lost"));if e!=nil || r.Dropped!=4 {t.Fatalf("drop control: %+v %v",r,e)}
 c.Close()
 for range CaptureRecordCount {l,e:=c.Next(context.Background());if e!=nil {t.Fatal(e)};if e=c.Done(l,nil);e!=nil {t.Fatal(e)}}
 l,e:=c.Next(context.Background());if e!=nil {t.Fatal(e)}
 got:=l.Record();defer c.Done(l,nil)
 if got.Offset!=12 || got.LostBefore!=4 || len(got.Data)!=0 {t.Fatalf("lost [8,12) must be represented before offset 12, got offset=%d lost=%d data=%d",got.Offset,got.LostBefore,len(got.Data))}
}
func TestRootCanceledNextDoesNotTakeQueuedRecord(t *testing.T) {
 c,e:=NewCapture("segment");if e!=nil {t.Fatal(e)}
 if _,e=c.Offer(Stdout,[]byte("owned"));e!=nil {t.Fatal(e)}
 ctx,cancel:=context.WithCancel(context.Background());cancel()
 l,e:=c.Next(ctx)
 if !errors.Is(e,context.Canceled) {t.Fatalf("canceled Next consumed record: err=%v record=%+v",e,l.Record())}
}

func TestRootCloseWaitsForPendingLossAfterAllSlotsLeased(t *testing.T) {
 c,e:=NewCapture("segment");if e!=nil {t.Fatal(e)}
 for range CaptureRecordCount {if _,e=c.Offer(Stdout,[]byte("x"));e!=nil {t.Fatal(e)}}
 var leased []Lease
 for range CaptureRecordCount {l,e:=c.Next(context.Background());if e!=nil {t.Fatal(e)};leased=append(leased,l)}
 if r,e:=c.Offer(Stdout,[]byte("lost"));e!=nil || r.Dropped!=4 {t.Fatalf("drop control %+v %v",r,e)}
 c.Close()
 ctx,cancel:=context.WithTimeout(context.Background(),20*time.Millisecond);defer cancel()
 _,e=c.Next(ctx)
 if !errors.Is(e,context.DeadlineExceeded) {t.Fatalf("closed before outstanding final loss could be queued: %v",e)}
 if e=c.Done(leased[0],nil);e!=nil {t.Fatal(e)}
 l,e:=c.Next(context.Background());if e!=nil {t.Fatal(e)}
 if r:=l.Record();r.Offset!=12 || r.LostBefore!=4 {t.Fatalf("late final loss: %+v",r)}
}
func TestRootCaptureOffsetOverflowFailsClosed(t *testing.T) {
 c,e:=NewCapture("segment");if e!=nil {t.Fatal(e)}
 c.offsets[int(Stdout)]=math.MaxUint64-1
 if _,e=c.Offer(Stdout,[]byte("xx"));e==nil {t.Fatal("stream offset silently wrapped")}
 if c.offsets[int(Stdout)]!=math.MaxUint64-1 {t.Fatal("failed Offer consumed overflowing input")}
}

func TestRootDonePreservesAppendFailure(t *testing.T) {
 c,e:=NewCapture("segment");if e!=nil {t.Fatal(e)}
 if _,e=c.Offer(Stdout,[]byte("x"));e!=nil {t.Fatal(e)}
 l,e:=c.Next(context.Background());if e!=nil {t.Fatal(e)}
 failure:=errors.New("disk failed")
 if e=c.Done(l,failure);!errors.Is(e,failure) {t.Fatalf("append failure hidden: %v",e)}
}
